1/**
2 * @file espnow_example_main.c
3 * @brief Sistema Vestible de Monitoreo - Nodo Maestro (Master Controller)
4 *
5 * Controlador central que orquesta la recolección de datos desde 2 nodos remotos vía ESP-NOW,
6 * sincronización de RTC (DS3231), lectura de sensor de pulso local (ADC), y logging en SD (SPI).
7 *
8 * **Hardware:**
9 * - Procesador: ESP32 DevKit V1 (Dual-core 240 MHz, WiFi + BT)
10 * - Almacenamiento: MicroSD 32GB vía SPI (MOSI-23, MISO-19, SCK-18, CS-5)
11 * - Reloj: DS3231 I2C (SDA-21, SCL-22, 0x68)
12 * - Sensor: Pulse oximetry on GPIO34 (ADC, 12-bit)
13 *
14 * **Protocolo Inalámbrico:**
15 * - ESP-NOW point-to-point de 2 nodos remotos: MPU (roll/pitch/accel/gyro) y TOUCH
16 * - Payload: 40 bytes (datos_mpu_touch_t)
17 * - Frecuencia de muestreo: configurable por nodo
18 *
19 * **Optimizaciones Energéticas:**
20 * - Dynamic Frequency Scaling: CPU 240 MHz → 80 MHz (activo) → 10 MHz (idle)
21 * - WiFi Power Save: WIFI_PS_NONE (mejor recepción/ACK para ESP-NOW)
22 * - Light Sleep con fallback (si no soportado, solo DFS)
23 * - TX power reducido a 10 dBm (40 unidades)
24 *
25 * **Filtrado de Señal:**
26 * - Pulso: Burst de 8 muestras → media recortada (sin min/max) → EMA(0.85) → baseline_ema(0.98) → remoción DC
27 * - Nodos IMU: validación de tamaño (40 bytes) + versión=1, timeout >1000ms limpia datos stale
28 *
29 * **Logging:**
30 * - CSV local buffering: 60 filas en RAM → escritura cada minuto a /sdcard/datos.csv
31 * - LIVE console: cada 10 segundos (reducido de 1s para ahorro de potencia)
32 * - Granularidad: 1 segundo por fila de registro
33 *
34 * **Framework:** ESP-IDF v6.0 (drivers modernos: esp_adc, esp_pm; legacy I2C driver still functional)
35 *
36 * @author Carlos Daniel Cuellar Antury
37 * @date 2026-04-05
38 * @version 1.2
39 */

146void imprimir_contenido_sd(void) { // Función para imprimir el contenido del archivo CSV en la consola
147 if (!sd_montada) {
148 ESP_LOGW(TAG, "SD no montada; no se puede leer datos.csv");
149 return;
150 }
151 FILE *f = fopen(SD_MOUNT_POINT"/datos.csv", "r"); // Abrir archivo en modo lectura
152 if (f) { // Si se abrió
153 // Mostrar solo las líneas recientes para no saturar el monitor
154 enum { MAX_RECIENTES = 20, MAX_LINEA = 192 };
155 static char linea[MAX_LINEA];
156 static char recientes[MAX_RECIENTES][MAX_LINEA];
157 int total = 0;
158 int pos = 0;
159
160 while (fgets(linea, sizeof(linea), f)) {
161 strncpy(recientes[pos], linea, MAX_LINEA - 1);
162 recientes[pos][MAX_LINEA - 1] = '\0';
163 pos = (pos + 1) % MAX_RECIENTES;
164 if (total < MAX_RECIENTES) {
165 total++;
166 }
167 }
168
169 ESP_LOGI(TAG, "Ultimas %d lineas de datos.csv:", total);
170 int inicio = (total == MAX_RECIENTES) ? pos : 0;
171 for (int i = 0; i < total; i++) {
172 int idx = (inicio + i) % MAX_RECIENTES;
173 ESP_LOGI(TAG, "%s", recientes[idx]);
174 }
175 fclose(f); // Cerrar archivo
176 } else {
177 ESP_LOGE(TAG, "No se pudo abrir el archivo para lectura"); // Error si no se abre
178 }
179}

181/**
182 * @brief Calcula los ángulos de roll y pitch a partir de datos de aceleración
183 *
184 * Utiliza una aproximación con arcotangente basada en los vectores de aceleración.
185 * Útil para casos donde se requiere estimación rápida sin giroscopio o como validación.
186 *
187 * @param acc_x Aceleración en eje X [g]
188 * @param acc_y Aceleración en eje Y [g]
189 * @param acc_z Aceleración en eje Z [g]
190 * @param[out] roll Puntero donde se almacena el ángulo roll calculado [grados]
191 * @param[out] pitch Puntero donde se almacena el ángulo pitch calculado [grados]
192 *
193 * @return void
194 *
195 * @details
196 * - Roll: atan2(acc_y, acc_z) * 180/π
197 * - Pitch: atan2(-acc_x, sqrt(acc_y² + acc_z²)) * 180/π
198 * - Rango: ±90 grados en ambos ejes
199 * - No requiere calibración, pero sensible a vibraciones
200 */
201static void calcular_roll_pitch(float acc_x, float acc_y, float acc_z, float *roll, float *pitch) { // Aproximación usando aceleración
202 *roll = atan2f(acc_y, acc_z) * (180.0f / 3.14159265f);
203 *pitch = atan2f(-acc_x, sqrtf((acc_y * acc_y) + (acc_z * acc_z))) * (180.0f / 3.14159265f);
204}

271
272 esp_err_t ret_bus = spi_bus_initialize(host.slot, &bus_cfg, SDSPI_DEFAULT_DMA); // Inicializar bus SPI
273 if (ret_bus != ESP_OK && ret_bus != ESP_ERR_INVALID_STATE) {

286 slot_config.gpio_cs = PIN_CS_SD;
287 slot_config.gpio_cd = SDSPI_SLOT_NO_CD;

289 slot_config.host_id = host.slot;

299 ESP_LOGI(TAG, "Tarjeta SD montada correctamente en %s", SD_MOUNT_POINT);

302// Lectura manual de registros BCD del DS3231
303/**
304 * @brief Lee la fecha y hora actual del chip RTC DS3231 vía I2C
305 *
306 * Realiza lectura secuencial de 7 registros BCD (0x00–0x06) del DS3231:
307 * - [0] Segundos (CH bit en bit[7] ignorado)
308 * - [1] Minutos
309 * - [2] Horas (24h)
310 * - [3] Día de semana (no utilizado)
311 * - [4] Día del mes
312 * - [5] Mes (Century bit en bit[7] ignorado)
313 * - [6] Año (0-99, interpretado como 2000-2099)
314 *
315 * Todos los valores se convierten de BCD a decimal y almacenan en struct tm.
316 * La lectura solo se acepta si rtc_tiempo_valido() pasa, caso contrario se conserva valor previo.
317 *
318 * @param[out] info_tiempo Puntero a struct tm donde se almacena la fecha/hora decodificada
319 *
320 * @return true si lectura exitosa y datos válidos; false si error I2C o validación fallida
321 *
322 * @pre I2C debe estar inicializado (@ref init_rtc_i2c fue llamado)
323 * @post Si retorna true, @c info_tiempo contiene fecha/hora válida
324 * @post Si retorna false, @c info_tiempo se conserva sin cambios (último valor válido)
325 *
326 * @note Timeout I2C: 1000 ms (@c portTICK_PERIOD_MS)
327 * @see rtc_escribir_tiempo para escritura; inicialmente RTC usa hora de compilación como fallback
328 *
329 * @author Carlos Daniel Cuellar Antury
330 */
331bool rtc_leer_tiempo(struct tm *info_tiempo) {
332 uint8_t datos[7]; // Búfer para datos de tiempo - Array para 7 bytes de datos
333 esp_err_t ret = ESP_FAIL;
334
335 // Reintentos para tolerar timeout I2C esporádico (ruido/carga de CPU).
336 for (int intento = 0; intento < 3; intento++) {
337 i2c_cmd_handle_t cmd = i2c_cmd_link_create(); // Crear comando I2C - Crea handle de comando
338 i2c_master_start(cmd); // Iniciar comando - Comando de start
339 i2c_master_write_byte(cmd, (DS3231_ADDR << 1) | I2C_MASTER_WRITE, true); // Escribir dirección - Dirección de escritura
340 i2c_master_write_byte(cmd, 0x00, true); // Dirección de registro - Registro 0x00 (segundos)
341 i2c_master_start(cmd); // Inicio repetido - Start repetido para lectura
342 i2c_master_write_byte(cmd, (DS3231_ADDR << 1) | I2C_MASTER_READ, true); // Dirección de lectura - Dirección de lectura
343 for (int i = 0; i < 6; i++) i2c_master_read_byte(cmd, &datos[i], I2C_MASTER_ACK); // Leer 6 bytes con ACK
344 i2c_master_read_byte(cmd, &datos[6], I2C_MASTER_NACK); // Último byte con NACK
345 i2c_master_stop(cmd); // Detener comando - Comando de stop
346 ret = i2c_master_cmd_begin(I2C_PORT_NUM, cmd, 1000 / portTICK_PERIOD_MS); // Ejecutar comando - Envía el comando
347 i2c_cmd_link_delete(cmd); // Eliminar comando - Libera el handle
348
349 if (ret == ESP_OK) {
350 break;
351 }
352 vTaskDelay(pdMS_TO_TICKS(10));
353 }
354
355 if (ret != ESP_OK) {
356 ESP_LOGW(TAG, "Lectura RTC falló tras reintentos: %s", esp_err_to_name(ret));
357 return false;
358 }
359 // Enmascarar bits de control del DS3231 antes de convertir desde BCD
360 struct tm nuevo = {0};
361 uint8_t sec_reg = (uint8_t)(datos[0] & 0x7F); // CH en bit7
362 uint8_t min_reg = (uint8_t)(datos[1] & 0x7F);
363 uint8_t hour_reg = (uint8_t)(datos[2] & 0x3F); // 24h
364 uint8_t mday_reg = (uint8_t)(datos[4] & 0x3F);
365 uint8_t mon_reg = (uint8_t)(datos[5] & 0x1F); // century en bit7
366 uint8_t year_reg = datos[6];
367
368 nuevo.tm_sec = bcd_a_dec(sec_reg);
369 nuevo.tm_min = bcd_a_dec(min_reg);
370 nuevo.tm_hour = bcd_a_dec(hour_reg);
371 nuevo.tm_mday = bcd_a_dec(mday_reg);
372 nuevo.tm_mon = (int)bcd_a_dec(mon_reg) - 1;
373 nuevo.tm_year = (int)bcd_a_dec(year_reg) + 100; // base 2000
374
375 if (rtc_tiempo_valido(&nuevo)) {
376 *info_tiempo = nuevo; // Solo actualizar si el tiempo leído es consistente
377 return true;
378 } else {
379 ESP_LOGW(TAG, "RTC devolvio tiempo invalido, se conserva ultimo valor valido");
380 return false;
381 }
382}

384/**
385 * @brief Escribe fecha y hora completa en el chip RTC DS3231 vía I2C
386 *
387 * Convierte struct tm a formato BCD y transmite 7 bytes al DS3231 (registros 0x00–0x06):
388 * - [0] Segundos (tm_sec)
389 * - [1] Minutos (tm_min)
390 * - [2] Horas (tm_hour, formato 24h)
391 * - [3] Día de semana (fijo a 1, no utilizado)
392 * - [4] Día del mes (tm_mday)
393 * - [5] Mes (tm_mon + 1, rango 1-12)
394 * - [6] Año (tm_year + 1900, modulo 100 para obtener 0-99)
395 *
396 * Típicamente se llama una sola vez al arranque, después mediante comando SETTIME por consola.
397 *
398 * @param t Puntero a struct tm con fecha/hora a escribir
399 *
400 * @return void
401 *
402 * @pre I2C debe estar inicializado (@ref init_rtc_i2c fue llamado)
403 * @pre Entrada @c t debe contener valores válidos según @ref rtc_tiempo_valido
404 * @post Si exitoso, RTC mantiene la hora (batería respaldada, persiste entre resets)
405 *
406 * **Lado efectos:**
407 * - Log INFO si éxito
408 * - Log ERROR si fallo I2C
409 *
410 * @note Timeout I2C: 1000 ms
411 * @see rtc_leer_tiempo para lectura; parsear_settime para parser del comando SETTIME
412 *
413 * @author Carlos Daniel Cuellar Antury
414 */
415void rtc_escribir_tiempo(const struct tm *t) { // Escribe fecha/hora completa en DS3231
416 uint8_t datos[7];
417 datos[0] = dec_a_bcd((uint8_t)t->tm_sec); // segundos
418 datos[1] = dec_a_bcd((uint8_t)t->tm_min); // minutos
419 datos[2] = dec_a_bcd((uint8_t)t->tm_hour); // horas (24h)
420 datos[3] = dec_a_bcd(1); // día semana fijo
421 datos[4] = dec_a_bcd((uint8_t)t->tm_mday); // día mes
422 datos[5] = dec_a_bcd((uint8_t)(t->tm_mon + 1)); // mes (1..12)
423 datos[6] = dec_a_bcd((uint8_t)((t->tm_year + 1900) % 100)); // año (00..99)
424
425 i2c_cmd_handle_t cmd = i2c_cmd_link_create();
426 i2c_master_start(cmd);
427 i2c_master_write_byte(cmd, (DS3231_ADDR << 1) | I2C_MASTER_WRITE, true);
428 i2c_master_write_byte(cmd, 0x00, true); // Registro inicial segundos
429 i2c_master_write(cmd, datos, sizeof(datos), true);
430 i2c_master_stop(cmd);
431 esp_err_t ret = i2c_master_cmd_begin(I2C_PORT_NUM, cmd, 1000 / portTICK_PERIOD_MS);
432 i2c_cmd_link_delete(cmd);
433
434 if (ret == ESP_OK) {
435 ESP_LOGI(TAG, "RTC actualizado: %02d/%02d/%04d %02d:%02d:%02d",
436 t->tm_mday, t->tm_mon + 1, t->tm_year + 1900,
437 t->tm_hour, t->tm_min, t->tm_sec);
438 } else {
439 ESP_LOGE(TAG, "No se pudo escribir RTC: %s", esp_err_to_name(ret));
440 }
441}

443/**
444 * @brief Convierte un valor decimal (0-99) a formato Binary Coded Decimal (BCD)
445 *
446 * El chip RTC DS3231 almacena fecha/hora en BCD: cada dígito decimal se codifica
447 * en 4 bits (nibble). Por ejemplo, 42 decimal → 0x42 BCD (4 en bits [7:4], 2 en bits [3:0]).
448 *
449 * @param val Valor decimal en rango [0, 99]
450 *
451 * @return Byte en formato BCD (dos dígitos decimales empaquetados)
452 *
453 * @example dec_a_bcd(59) → (5 << 4) | 9 = 0x59
454 * @see bcd_a_dec para conversión inversa
455 *
456 * @author Carlos Daniel Cuellar Antury
457 */
458static uint8_t dec_a_bcd(uint8_t val) { // Convertir decimal a BCD para DS3231
459 return (uint8_t)(((val / 10) << 4) | (val % 10));
460}
461
462/**
463 * @brief Convierte un byte en formato Binary Coded Decimal (BCD) a decimal
464 *
465 * Operación inversa de dec_a_bcd(). Extrae los dos dígitos BCD y los convierte a valor decimal.
466 *
467 * @param val Byte en formato BCD (rango [0x00, 0x99])
468 *
469 * @return Valor decimal en rango [0, 99]
470 *
471 * @example bcd_a_dec(0x42) → (4 * 10) + 2 = 42
472 * @see dec_a_bcd para conversión inversa
473 *
474 * @author Carlos Daniel Cuellar Antury
475 */
476static uint8_t bcd_a_dec(uint8_t val) { // Convertir BCD a decimal
477 return (uint8_t)(((val >> 4) * 10) + (val & 0x0F));
478}

518/**
519 * @brief Valida rangos de una estructura de tiempo (struct tm) del RTC
520 *
521 * Comprueba que todos los campos de fecha/hora se encuentren dentro de rangos válidos.
522 * Útil para detectar lecturas corruptas del DS3231 o BCD conversiones errónicas.
523 *
524 * **Rangos validados:**
525 * - Segundo: [0, 59]
526 * - Minuto: [0, 59]
527 * - Hora: [0, 23]
528 * - Día del mes: [1, 31]
529 * - Mes: [0, 11] (tm_mon es 0-indexed: enero=0, diciembre=11)
530 * - Año: [100, 199] (tm_year es años desde 1900, rango [2000, 2099])
531 *
532 * @param t Puntero a estructura tm con datos a validar
533 *
534 * @return true si todos los campos están en rango; false si alguno está fuera
535 *
536 * @note **IMPORTANTE:** tm_mon es 0-indexed. Enero = 0, diciembre = 11. Se validó esta corrección en v1.2.
537 * @see rtc_leer_tiempo, rtc_escribir_tiempo para operaciones de lectura/escritura del DS3231
538 *
539 * @author Carlos Daniel Cuellar Antury
540 */
541static bool rtc_tiempo_valido(const struct tm *t) { // Validar rangos básicos del tiempo leído
542 if (t->tm_sec < 0 || t->tm_sec > 59) return false;
543 if (t->tm_min < 0 || t->tm_min > 59) return false;
544 if (t->tm_hour < 0 || t->tm_hour > 23) return false;
545 if (t->tm_mday < 1 || t->tm_mday > 31) return false;
546 if (t->tm_mon < 0 || t->tm_mon > 11) return false;
547 if (t->tm_year < 100 || t->tm_year > 199) return false; // 2000..2099
548 return true;
549}

551/**
552 * @brief Parser del comando de sincronización RTC vía consola serial: "SETTIME YYYY-MM-DD HH:MM:SS"
553 *
554 * Decodifica una línea de texto recibida por UART/ConsoleI O y extrae campos de fecha/hora.
555 * Valida el comando contra formato exacto y rango de valores. Si parse y validación éxitos,
556 * copia a estructura tm de salida.
557 *
558 * **Formato esperado:**
559 * ```
560 * SETTIME 2026-04-05 14:30:45
561 * ↑ ↑ ↑ ↑ ↑ ↑ ↑ ↑
562 * cmd Y M D H M S
563 * ```
564 *
565 * **Validación:**
566 * - Exactamente 6 campos numéricos extraídos por sscanf
567 * - Rango de valores chequea via @ref rtc_tiempo_valido
568 * - Si falla parsing o validación: retorna false sin modificar @c out
569 *
570 * @param linea Línea de texto (presumiblemente de fgets, puede incluir \\n)
571 * @param[out] out Puntero a struct tm donde se copia fecha/hora si éxito
572 *
573 * @return true si parsing y validación exitosos; false en caso contrario
574 *
575 * **Conversión:**
576 * - Año ISO (4 dígitos) → tm_year = año - 1900
577 * - Mes ISO (1-12) → tm_mon = mes - 1 (0-indexed)
578 * - Resto campos copiados directamente
579 *
580 * @note Consumo: ~50 bytes stack (struct tm local)
581 * @see task_settime_serial para loop de lectura de consola
582 * @see rtc_escribir_tiempo para escritura del RTC después de parse exitoso
583 *
584 * @author Carlos Daniel Cuellar Antury
585 */
586static bool parsear_settime(const char *linea, struct tm *out) { // Parsear: SETTIME YYYY-MM-DD HH:MM:SS
587 int anio, mes, dia, hora, minuto, segundo;
588 if (sscanf(linea, "SETTIME %d-%d-%d %d:%d:%d", &anio, &mes, &dia, &hora, &minuto, &segundo) != 6) {
589 return false;
590 }
591 struct tm t = {0};
592 t.tm_year = anio - 1900;
593 t.tm_mon = mes - 1;
594 t.tm_mday = dia;
595 t.tm_hour = hora;
596 t.tm_min = minuto;
597 t.tm_sec = segundo;
598 if (!rtc_tiempo_valido(&t)) {
599 return false;
600 }
601 *out = t;
602 return true;
603}

627/**
628 * @brief Callback de recepción del protocolo ESP-NOW
629 *
630 * Manejador de interrupciones ejecutado cuando llega un paquete ESP-NOW de uno de los nodos remotos.
631 * Valida MAC, tamaño de payload y sincroniza datos en la estructura global @ref datos_sistema con protección de Mutex.
632 *
633 * **Validación:**
634 * - MAC: Compara contra lista conocida (MPU: 68:25:dd:32:70:cc, TOUCH: b4:3a:45:26:0e:40)
635 * - Tamaño: Rechaza si no es exactamente 40 bytes (@c sizeof(datos_mpu_touch_t))
636 * - Versión: Rechaza si @c version != 1
637 * - Si MAC desconocida o tamaño incorrecto: log de advertencia, datos no copiados
638 *
639 * **Thread-safety:**
640 * - Toma @ref xMutex antes de escribir @ref datos_sistema
641 * - Actualiza timestamp en @ref ultimo_rx_mpu_ms o @ref ultimo_rx_touch_ms
642 * - Incrementa contador @ref paquetes_mpu o @ref paquetes_touch
643 *
644 * @param info Información de emisor (MAC source, RSSI, channel)
645 * @param data Payload (esperado: datos_mpu_touch_t = 40 bytes)
646 * @param len Longitud actual del payload [bytes]
647 *
648 * @return void
649 *
650 * @warning Si @ref xMutex es NULL: función retorna sin procesar (deadlock prevention)
651 * @warning touch_raw/touch_filtrado en cero son válidos (p.ej. en slave MPU)
652 *
653 * @author Carlos Daniel Cuellar Antury
654 */
655void cb_recepcion(const esp_now_recv_info_t *info, const uint8_t *data, int len) { // Callback para recepción ESP-NOW
656 if (xMutex == NULL || info == NULL || data == NULL) {
657 return;
658 }
659 if (len != (int)sizeof(datos_mpu_touch_t)) {
660 ESP_LOGW(TAG, "ESP-NOW tamaño invalido MAC %02x:%02x:%02x:%02x:%02x:%02x len=%d esperado=%d",
661 info->src_addr[0], info->src_addr[1], info->src_addr[2],
662 info->src_addr[3], info->src_addr[4], info->src_addr[5],
663 len, (int)sizeof(datos_mpu_touch_t));
664 return;
665 }
666
667 datos_mpu_touch_t rx;
668 memcpy(&rx, data, sizeof(rx));
669 if (rx.version != 1) {
670 ESP_LOGW(TAG, "ESP-NOW version invalida MAC %02x:%02x:%02x:%02x:%02x:%02x version=%u esperado=1",
671 info->src_addr[0], info->src_addr[1], info->src_addr[2],
672 info->src_addr[3], info->src_addr[4], info->src_addr[5],
673 (unsigned)rx.version);
674 return;
675 }
676
677 int64_t ahora_ms = (int64_t)esp_log_timestamp();
678 const bool es_mpu = (memcmp(info->src_addr, mac_esclavo_mpu, ESP_NOW_ETH_ALEN) == 0);
679 const bool es_touch = (memcmp(info->src_addr, mac_esclavo_touch, ESP_NOW_ETH_ALEN) == 0);
680 if (!es_mpu && !es_touch) {
681 ESP_LOGW(TAG, "MAC desconocida: %02x:%02x:%02x:%02x:%02x:%02x len=%d",
682 info->src_addr[0], info->src_addr[1], info->src_addr[2],
683 info->src_addr[3], info->src_addr[4], info->src_addr[5], len);
684 return;
685 }
686
687 bool imprimir_rx = false;
688 if (es_mpu) {
689 if (ultimo_log_rx_mpu_ms < 0 || (ahora_ms - ultimo_log_rx_mpu_ms) >= RX_LOG_INTERVAL_MS) {
690 imprimir_rx = true;
691 ultimo_log_rx_mpu_ms = ahora_ms;
692 }
693 } else if (es_touch) {
694 if (ultimo_log_rx_touch_ms < 0 || (ahora_ms - ultimo_log_rx_touch_ms) >= RX_LOG_INTERVAL_MS) {
695 imprimir_rx = true;
696 ultimo_log_rx_touch_ms = ahora_ms;
697 }
698 }
699
700 xSemaphoreTake(xMutex, portMAX_DELAY); // Bloquear acceso - Toma el mutex
701 // Identificar quién envió el paquete comparando la MAC
702 if (es_mpu) {
703 datos_sistema.mpu = rx; // Guardar para uso de aplicación/alertas
704 paquetes_mpu++;
705 ultimo_rx_mpu_ms = ahora_ms;
706 } else if (es_touch) {
707 datos_sistema.touch = rx; // Guardar para uso de aplicación/alertas
708 paquetes_touch++;
709 ultimo_rx_touch_ms = ahora_ms;
710 }
711 xSemaphoreGive(xMutex); // Liberar acceso - Libera el mutex
712
713 if (imprimir_rx) {
714 ESP_LOGI(TAG,
715 "RX %s %02x:%02x:%02x:%02x:%02x:%02x | roll=%.2f pitch=%.2f | ACC(%.2f,%.2f,%.2f) | GYR(%.2f,%.2f,%.2f) | estado=%u(%s)",
716 es_mpu ? "MPU" : "TOUCH",
717 info->src_addr[0], info->src_addr[1], info->src_addr[2],
718 info->src_addr[3], info->src_addr[4], info->src_addr[5],
719 rx.roll, rx.pitch,
720 rx.acc_x, rx.acc_y, rx.acc_z,
721 rx.gyr_x, rx.gyr_y, rx.gyr_z,
722 (unsigned)rx.estado, estado_a_texto(rx.estado));
723 }
724}

728/**
729 * @brief Tarea de lectura periódica del sensor de pulso con filtrado robusto
730 *
731 * Lee el sensor de pulso cada 500ms usando ADC1 canal 6 (GPIO34) con resolución 12-bit.
732 * Implementa tres etapas de filtrado para extraer la amplitud AC (latidos) removiendo ruido y DC bias:
733 *
734 * **Etapa 1: Media recortada (trimmed mean)**
735 * - Adquiere 8 muestras consecutivas
736 * - Elimina mínimo y máximo (outliers espurios)
737 * - Promedia las 6 muestras restantes
738 *
739 * **Etapa 2: EMA rápida (0.85)**
740 * - Suaviza fluctuaciones de corta duración
741 * - Nuevo valor = 0.85 × anterior + 0.15 × actual
742 *
743 * **Etapa 3: Baseline lenta (0.98) + Remoción DC**
744 * - Baseline_EMA sigue drift lento del contacto (factor 0.98)
745 * - Compone AC = señal_EMA - baseline_EMA
746 * - Escala visible: |AC| × 6.0 (±2047 rango típico)
747 *
748 * **Banda muerta:** ±5 LSB para evitar parpadeo
749 * **Consumo:** ~85 mA (ADC continuous + mutex contention despreciable)
750 * **Publicación:** Cada ciclo (500 ms) en @ref datos_sistema.pulso_raw
751 *
752 * @param p Parámetro de tarea FreeRTOS (no utilizado)
753 *
754 * @return void (tarea infinita)
755 *
756 * @thread Thread-safe: protege escritura en @ref datos_sistema con @ref xMutex
757 * @note Inicializa ADC1 en primer ciclo; no requiere init_adc_pulso() previo
758 * @see calcular_roll_pitch para estimación de ángulos desde aceleración
759 *
760 * @author Carlos Daniel Cuellar Antury
761 */
762void task_pulso(void *p) { // Tarea para leer sensor de pulso
763 // Configuración del ADC Unidad 1
764 adc_oneshot_unit_init_cfg_t init_config1 = { .unit_id = ADC_UNIT_1 }; // Config inicial ADC
765 ESP_ERROR_CHECK(adc_oneshot_new_unit(&init_config1, &adc1_handle)); // Crear unidad ADC - Inicializa unidad ADC
766 // Configuración del Canal del Sensor
767 adc_oneshot_chan_cfg_t config = { // Config del canal
768 .bitwidth = ADC_BITWIDTH_12, // 12 bits - Resolución
769 .atten = ADC_ATTEN_DB_12, // Atenuación - Atenuación para rango
770 };
771 ESP_ERROR_CHECK(adc_oneshot_config_channel(adc1_handle, CANAL_LATIDOS, &config)); // Configurar canal - Configura el canal ADC
772
773 // Filtro robusto y extracción de amplitud (quita nivel DC ~1800 típico del sensor).
774 enum { PULSO_MUESTRAS_BURST = 8 };
775 float pulso_ema = -1.0f;
776 float baseline_ema = -1.0f;
777 int pulso_prev = 0;
778
779 while(1) { // Bucle infinito
780 int suma = 0;
781 int min_v = 4095;
782 int max_v = 0;
783
784 for (int i = 0; i < PULSO_MUESTRAS_BURST; i++) {
785 int adc_raw = 0;
786 ESP_ERROR_CHECK(adc_oneshot_read(adc1_handle, CANAL_LATIDOS, &adc_raw)); // Leer ADC
787 suma += adc_raw;
788 if (adc_raw < min_v) min_v = adc_raw;
789 if (adc_raw > max_v) max_v = adc_raw;
790 }
791
792 // Promedio recortado: quita un mínimo y un máximo para reducir picos espurios.
793 int promedio_recortado = (suma - min_v - max_v) / (PULSO_MUESTRAS_BURST - 2);
794 if (pulso_ema < 0.0f) {
795 pulso_ema = (float)promedio_recortado;
796 } else {
797 pulso_ema = (0.85f * pulso_ema) + (0.15f * (float)promedio_recortado);
798 }
799
800 // Baseline lento para seguir deriva de contacto sin comerse el latido.
801 if (baseline_ema < 0.0f) {
802 baseline_ema = pulso_ema;
803 } else {
804 baseline_ema = (0.98f * baseline_ema) + (0.02f * pulso_ema);
805 }
806
807 float pulso_ac = pulso_ema - baseline_ema; // Componente útil (AC)
808 int pulso_filtrado = (int)(fabsf(pulso_ac) * 2.0f + 0.5f); // Escala calibrada para HR local
809 if (pulso_filtrado > 4095) {
810 pulso_filtrado = 4095;
811 }
812 if (abs(pulso_filtrado - pulso_prev) < 3) {
813 pulso_filtrado = pulso_prev; // Banda muerta pequeña para quitar parpadeo
814 }
815 pulso_prev = pulso_filtrado;
816
817 xSemaphoreTake(xMutex, portMAX_DELAY); // Tomar mutex - Protege datos
818 datos_sistema.pulso_raw = pulso_filtrado; // Publicar señal estabilizada
819 xSemaphoreGive(xMutex); // Dar mutex - Libera mutex
820
821 int64_t ahora_ms = (int64_t)esp_log_timestamp();
822 if (ultimo_log_hr_ms < 0 || (ahora_ms - ultimo_log_hr_ms) >= HR_LOG_INTERVAL_MS) {
823 ultimo_log_hr_ms = ahora_ms;
824 ESP_LOGI(TAG, "HR local=%d", pulso_filtrado);
825 }
826
827 vTaskDelay(pdMS_TO_TICKS(500)); // Muestreo cada 500ms - Ahorro batería
828 }
829}

831/**
832 * @brief Tarea de logging en SD con buffering y detección de desconexión
833 *
834 * Recolecta datos sincronizados (pulso local + IMU remota + RTC) cada 1 segundo, acumula en buffer RAM,
835 * y escribe a /sdcard/datos.csv cada ~60 segundos (cuando buffer lleno). Detecta timeout de nodos
836 * remotos >1000ms y limpia datos residuales para evitar "ghost data" en los registros.
837 *
838 * **Ciclo de operación:**
839 * 1. Delay 1000 ms
840 * 2. Leer snapshot de @ref datos_sistema (con protección @ref xMutex)
841 * 3. Validar conectividad: si último_rx_ms > RX_TIMEOUT_MS, marcar desconectado + memset zeros
842 * 4. Generar línea CSV en lote[n] (~192 bytes)
843 * 5. Si lote lleno (60 líneas): apertura archivo, escribir búfer, cerrar (atomic write)
844 * 6. Log LIVE cada 10 segundos a consola (1 de cada 10 ciclos)
845 *
846 * **Formato CSV:**
847 * ```
848 * DD/MM/YYYY,HH:MM:SS,pulso_raw,mpu_roll,mpu_pitch,mpu_accXYZ[3],mpu_gyrXYZ[3],
849 * touch_roll,touch_pitch,touch_accXYZ[3],touch_gyrXYZ[3]
850 * ```
851 * Total: 23 campos por fila = ~192 bytes (sin comillas)
852 *
853 * **Optimizaciones energéticas:**
854 * - SD write: 1 por minuto (60 rows), no 60 writes por minuto → ~40 mA ahorro
855 * - LIVE console: cada 10 segundos (no cada 1s) → UART duty cycle 90% menos
856 * - Timeout cleanup: memset(0) para nodos desconectados evita registros fantasma
857 *
858 * **Monitoreo de conectividad:**
859 * - Última recepción: último_rx_mpu_ms, último_rx_touch_ms (actualizados en @ref cb_recepcion)
860 * - Timeout: >1000 ms sin paquete → log de desconexión + cero en CSV
861 * - Reconexión: automática detectada, log de notificación
862 *
863 * @param p Parámetro de tarea FreeRTOS (no utilizado)
864 *
865 * @return void (tarea infinita)
866 *
867 * @pre SD debe estar montada (@ref sd_montada = true), sino datos se bufferean sin escribir
868 * @thread Thread-safe: toma @ref xMutex solo para leer snapshot de @ref datos_sistema, libera inmediato
869 * @post Archivo /sdcard/datos.csv crecerá ~11 KB cada minuto
870 *
871 * @warning Si SD no está disponible: los datos se pierden (no hay retry buffer persistente)
872 * @note Granularidad: 1 segundo por fila, ~1 minuto entre writes
873 * @see imprimir_contenido_sd para lectura y visualización de datos guardados
874 *
875 * @author Carlos Daniel Cuellar Antury
876 */
877void task_sd_log(void *p) { // Tarea para guardar en SD - lote cada 60s para ahorrar batería
878 enum { LOTE_MAX = 60, MAX_CSV = 256 };
879 static char lote[LOTE_MAX][MAX_CSV]; // Búfer estático ~11KB; fuera del stack
880 int n = 0;
881 int live_cnt = 0;
882 while(1) {
883 vTaskDelay(pdMS_TO_TICKS(1000)); // Muestreo cada 1s
884 int64_t ahora_ms = (int64_t)esp_log_timestamp();
885 log_completo_t snap;
886 uint32_t cnt_mpu;
887 uint32_t cnt_touch;
888 bool mpu_reciente;
889 bool touch_reciente;
890 xSemaphoreTake(xMutex, portMAX_DELAY);
891 snap = datos_sistema;
892 cnt_mpu = paquetes_mpu;
893 cnt_touch = paquetes_touch;
894 mpu_reciente = (ultimo_rx_mpu_ms >= 0) && ((ahora_ms - ultimo_rx_mpu_ms) <= RX_TIMEOUT_MS);
895 touch_reciente = (ultimo_rx_touch_ms >= 0) && ((ahora_ms - ultimo_rx_touch_ms) <= RX_TIMEOUT_MS);
896 xSemaphoreGive(xMutex);
897
898 if (!mpu_reciente && !mpu_desconectado) {
899 mpu_desconectado = true;
900 ESP_LOGW(TAG, "MPU sin paquetes > %d ms (contador=%lu)", RX_TIMEOUT_MS, (unsigned long)cnt_mpu);
901 } else if (mpu_reciente && mpu_desconectado) {
902 mpu_desconectado = false;
903 ESP_LOGI(TAG, "MPU reconectado (contador=%lu)", (unsigned long)cnt_mpu);
904 }
905 if (!touch_reciente && !touch_desconectado) {
906 touch_desconectado = true;
