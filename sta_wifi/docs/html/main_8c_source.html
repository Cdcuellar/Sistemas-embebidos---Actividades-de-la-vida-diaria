59 return (c >= 'A' && c <= 'Z') ||
60 (c >= 'a' && c <= 'z') ||
61 (c >= '0' && c <= '9') ||

68static void url_encode(const char *src, char *dst, size_t dst_size)
69{
70 size_t di = 0;
71
72 if (dst_size == 0) {
73 return;
74 }
75
76 for (size_t si = 0; src[si] != '\0' && di + 1 < dst_size; si++) {
77 unsigned char ch = (unsigned char)src[si];
78
79 if (is_url_unreserved((char)ch)) {
80 dst[di++] = (char)ch;
81 } else if (ch == ' ') {
82 dst[di++] = '+';
83 } else {
84 if (di + 3 >= dst_size) {
85 break;
86 }
87
88 int n = snprintf(&dst[di], dst_size - di, "%%%02X", ch);
89 if (n != 3) {
90 break;
91 }
92 di += 3;
93 }
94 }
95
96 dst[di] = '\0';
97}

144static char *trim_whitespace(char *str)
145{
146 char *end; // puntero usado para recortar al final
147
148 while (*str != '\0' && isspace((unsigned char)*str)) { // mover inicio sobre espacios
149 str++; // siguiente caracter
150 } // fin while
151
152 if (*str == '\0') { // si queda vacia tras trim-izquierda
153 return str; // devolver puntero a cadena vacia
154 } // fin verificacion vacio
155
156 end = str + strlen(str) - 1; // apuntar al ultimo caracter valido
157 while (end > str && isspace((unsigned char)*end)) { // retroceder sobre espacios
158 end--; // caracter anterior
159 } // fin while
160
161 end[1] = '\0'; // cerrar cadena despues del ultimo no-espacio
162 return str; // devolver inicio recortado
163} // fin de funcion

173static bool extract_json_value(const char *payload, const char *key,
174 char *output, size_t output_size)
175{
176 const char *found = strstr(payload, key); // localizar clave en payload
177 if (found == NULL) { // clave no encontrada
178 return false; // extraccion fallida
179 } // fin verificacion de clave
180
181 const char *colon = strchr(found, ':'); // buscar separador ':'
182 if (colon == NULL) { // par clave/valor mal formado
183 return false; // extraccion fallida
184 } // fin verificacion de ':'
185
186 colon++; // mover despues de ':'
187 while (*colon != '\0' && isspace((unsigned char)*colon)) { // saltar espacios
188 colon++; // siguiente caracter
189 } // fin trim-izquierda del valor
190
191 if (*colon == '"') { // cadena JSON entre comillas
192 colon++; // saltar comilla de apertura
193 const char *end_quote = strchr(colon, '"'); // buscar comilla de cierre
194 if (end_quote == NULL) { // si no hay comilla de cierre
195 return false; // cadena JSON mal formada
196 } // fin verificacion de comilla
197 size_t length = end_quote - colon; // longitud del valor en caracteres
198 if (length >= output_size) { // prevenir overflow
199 length = output_size - 1; // truncar de forma segura
200 } // fin ajuste de longitud
201 memcpy(output, colon, length); // copiar bytes del valor
202 output[length] = '\0'; // terminar cadena de salida
203 return true; // extraccion exitosa
204 } // fin rama con comillas
205
206 const char *end = colon; // inicio de valor sin comillas
207 while (*end != '\0' && *end != ',' && *end != '}' && *end != '\n' && *end != '\r') { // tokens de parada
208 end++; // avanzar hasta token
209 } // fin while
210
211 size_t length = end - colon; // longitud cruda del valor
212 if (length >= output_size) { // prevenir overflow
213 length = output_size - 1; // truncar de forma segura
214 } // fin ajuste de longitud
215 memcpy(output, colon, length); // copiar valor crudo
216 output[length] = '\0'; // terminar cadena de salida
217 trim_whitespace(output); // quitar espacios extra
218 return true; // extraccion exitosa
219} // fin de funcion

225static void procesar_respuesta(const char *payload)
226{
227 char valor[128] = {0}; // buffer para "valor"
228
229 if (extract_json_value(payload, "\"valor\"", valor, sizeof(valor))) { // intentar formato JSON
230 ESP_LOGI(TAG, "Extracted value (valor): %s", valor); // imprimir valor
231 return; // detener si parseo JSON fue exitoso
232 } // fin intento JSON
233
234 const char *dato_ptr = strstr(payload, "dato="); // intentar formato texto plano
235 if (dato_ptr != NULL) { // si existe el token
236 dato_ptr += strlen("dato="); // saltar prefijo del token
237 char dato[128] = {0}; // buffer para valor plano
238 const char *end = strchr(dato_ptr, '\n'); // buscar fin de linea
239 if (end == NULL) { // si no hay salto de linea
240 end = dato_ptr + strlen(dato_ptr); // usar fin de cadena
241 } // fin verificacion de linea
242 size_t length = end - dato_ptr; // longitud del valor
243 if (length >= sizeof(dato)) { // prevenir overflow
244 length = sizeof(dato) - 1; // truncar de forma segura
245 } // fin ajuste de longitud
246 memcpy(dato, dato_ptr, length); // copiar valor
247 dato[length] = '\0'; // terminar cadena
248 trim_whitespace(dato); // quitar espacios
249 ESP_LOGI(TAG, "Extracted value (dato=): %s", dato); // imprimir valor
250 return; // detener despues de parsear
251 } // fin rama de texto plano
252
253 ESP_LOGW(TAG, "No useful data pattern found in response."); // advertencia de fallback
254} // fin de funcion

274 xEventGroupSetBits(s_wifi_event_group, WIFI_CONNECTED_BIT); // marcar como conectado

346static esp_err_t send_http_request(void)
347{
348 char url[256]; // buffer de URL
349 char encoded_message[REQUEST_ENCODED_MAX_LEN]; // mensaje codificado para URL
350
351 url_encode(REQUEST_MESSAGE, encoded_message, sizeof(encoded_message)); // codificar query
352
353 snprintf(url, sizeof(url), "http://%s:%d%s?mensaje=%s", // construir URL final con mensaje
354 SERVER_HOST, SERVER_PORT, SERVER_PATH, encoded_message);
355 ESP_LOGI(TAG, "GET URL: %s", url); // imprimir URL
356
357 esp_http_client_config_t config = { // configuracion del cliente HTTP
358 .url = url, // endpoint URL
359 .method = HTTP_METHOD_GET, // solicitud GET
360 .timeout_ms = 10000, // timeout en milisegundos
361 };
362
363 esp_http_client_handle_t client = esp_http_client_init(&config); // crear handle de cliente
364 if (client == NULL) { // si falla reserva/inicializacion
365 ESP_LOGE(TAG, "Failed to initialize HTTP client"); // reportar error
366 return ESP_FAIL; // abortar llamada
367 } // fin verificacion de cliente
368
369 esp_err_t err = esp_http_client_open(client, 0); // abrir conexion y enviar headers
370 if (err == ESP_OK) { // si apertura fue exitosa
371 int content_length = esp_http_client_fetch_headers(client); // obtener headers de respuesta
372 int status_code = esp_http_client_get_status_code(client); // leer estado HTTP
373 ESP_LOGI(TAG, "HTTP status code: %d", status_code); // imprimir estado
374
375 if (status_code == 200) { // procesar solo 200 OK
376 char buffer[HTTP_BUFFER_SIZE + 1] = {0}; // buffer de payload + byte nulo
377 int total_read = 0; // bytes acumulados leidos
378
379 while (total_read < HTTP_BUFFER_SIZE) { // leer hasta llenar buffer o sin datos
380 int r = esp_http_client_read(client, buffer + total_read, HTTP_BUFFER_SIZE - total_read); // leer bloque
381 if (r <= 0) { // timeout/fin/error
382 break; // detener lectura
383 } // fin verificacion de bloque
384 total_read += r; // sumar bytes leidos
385 } // fin bucle de lectura
386
387 buffer[total_read] = '\0'; // forzar terminador nulo
388 ESP_LOGI(TAG, "Received content (%d bytes): %s", total_read, buffer); // imprimir payload
389 if (total_read > 0) { // si hay payload
390 procesar_respuesta(buffer); // parsear y loguear valor extraido
391 } else { // payload vacio
392 ESP_LOGW(TAG, "Empty response from server"); // advertencia
393 } // fin verificacion de payload
394
395 (void)content_length; // silenciar warning de no usado
396 } else { // respuesta distinta de 200
397 ESP_LOGW(TAG, "HTTP response not OK: %d", status_code); // advertencia
398 } // fin rama por estado
399 } else { // fallo apertura de conexion
400 ESP_LOGE(TAG, "HTTP request failed: %s", esp_err_to_name(err)); // imprimir codigo de error
401 } // fin rama de apertura
402
403 esp_http_client_cleanup(client); // liberar recursos del cliente HTTP
404 return err; // devolver resultado de operacion
405} // fin de funcion

410void app_main(void)
411{
412 esp_err_t ret = nvs_flash_init(); // inicializar flash NVS
413 if (ret == ESP_ERR_NVS_NO_FREE_PAGES || ret == ESP_ERR_NVS_NEW_VERSION_FOUND) { // NVS requiere borrado
414 ESP_ERROR_CHECK(nvs_flash_erase()); // borrar NVS invalido

416 } // fin recuperacion NVS
417 ESP_ERROR_CHECK(ret); // detener si NVS sigue fallando
418
419 ESP_LOGI(TAG, "Starting ESP-IDF STA HTTP client"); // log de arranque
420
421 wifi_init_sta(); // inicializar modo WiFi STA
422
423 while (true) { // bucle infinito de la tarea
424 if (is_wifi_connected()) { // si esta conectado a WiFi
425 ESP_LOGI(TAG, "Sending HTTP request..."); // log de inicio de solicitud
426 send_http_request(); // ejecutar una solicitud HTTP GET
427 } else { // si esta desconectado
428 ESP_LOGW(TAG, "Not connected to WiFi. Waiting for connection..."); // advertir desconexion
429 } // fin rama de conectividad
430
431 vTaskDelay(pdMS_TO_TICKS(HTTP_REQUEST_INTERVAL_MS)); // esperar antes del siguiente ciclo
432 } // fin bucle principal
433} // fin app_main
